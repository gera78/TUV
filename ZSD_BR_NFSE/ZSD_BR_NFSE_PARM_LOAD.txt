* Functional description                                              *
* One-off load of the proposed initial content of the parameter      *
* table ZSD_BR_NFSE_PARM for one company code (design document,       *
* section 8). Every proposal is classified as:                        *
*   - Defined: value known (control parameters, provider data of the *
*     output file, service descriptions, CFOP proposal)               *
*   - To be defined: value still depends on the tax consultant        *
*     (tax types, tax laws, item type, material, ...). These are      *
*     only loaded, with an EMPTY value, when "Include entries to be   *
*     defined" is ticked - the upload programs treat an existing      *
*     entry as maintained, so complete them in the maintenance dialog *
*     before the first real upload.                                   *
* Existing entries are kept unless "Overwrite existing values" is     *
* ticked. "Test run" (default) only lists what would be done.         *
* Entries are written directly to the table (no transport request):   *
* run it in each client, or transport the entries afterwards from the *
* maintenance dialog (SM30 - Table View > Transport).                 *
*---------------------------------------------------------------------*
* Authorization logic                                                 *
* S_TABU_NAM (ACTVT 02, table ZSD_BR_NFSE_PARM) - same authorization  *
* as the maintenance dialog of the table.                             *
*---------------------------------------------------------------------*
* Text symbols / selection texts                                      *
*   B01 Load options                                                  *
*   E01 Company code does not exist                                   *
*   E02 No authorization to change table ZSD_BR_NFSE_PARM             *
*   E03 Error writing ZSD_BR_NFSE_PARM - nothing saved                *
*   S01 entries written to ZSD_BR_NFSE_PARM                           *
*   P_BUKRS Company code       P_TBD   Include entries TBD            *
*   P_OVERWR Overwrite values  P_TEST  Test run                       *

*&---------------------------------------------------------------------*
*&      Form  CHECK_INPUT
*&---------------------------------------------------------------------*
FORM check_input.

  SELECT SINGLE bukrs FROM t001 WHERE bukrs = @p_bukrs INTO @DATA(lv_bukrs).
  IF sy-subrc <> 0.
    MESSAGE TEXT-e01 TYPE 'E'.                   " Company code does not exist
  ENDIF.

  AUTHORITY-CHECK OBJECT 'S_TABU_NAM'
    ID 'ACTVT'     FIELD '02'
    ID 'TABLE'     FIELD gc_table.
  IF sy-subrc <> 0.
    MESSAGE TEXT-e02 TYPE 'E'.                   " No authorization to change the table
  ENDIF.

ENDFORM.

*&      Form  PROCESS
*&---------------------------------------------------------------------*
FORM process.

  DATA: lt_existing TYPE SORTED TABLE OF zsd_br_nfse_parm
                         WITH UNIQUE KEY bukrs field_name file_column_name z_input,
        lt_write    TYPE STANDARD TABLE OF zsd_br_nfse_parm WITH EMPTY KEY,
        ls_out      TYPE gty_out,
        lv_count    TYPE char10.

  SELECT * FROM zsd_br_nfse_parm
    WHERE bukrs = @p_bukrs
    INTO TABLE @lt_existing.

  LOOP AT gt_prop INTO DATA(ls_prop).
    CLEAR ls_out.
    ls_out-field_name       = ls_prop-field_name.
    ls_out-file_column_name = ls_prop-file_column_name.
    ls_out-z_input          = ls_prop-z_input.
    ls_out-z_output         = ls_prop-z_output.
    ls_out-note             = ls_prop-note.

    READ TABLE lt_existing INTO DATA(ls_existing)
      WITH TABLE KEY bukrs            = p_bukrs
                     field_name       = ls_prop-field_name
                     file_column_name = ls_prop-file_column_name
                     z_input          = ls_prop-z_input.
    DATA(lv_exists) = xsdbool( sy-subrc = 0 ).
    IF lv_exists = abap_true.
      ls_out-old_output = ls_existing-z_output.
    ENDIF.

    IF ls_prop-tbd = abap_true AND p_tbd = abap_false.
      ls_out-icon   = icon_led_yellow.
      ls_out-action = 'Skipped - to be defined'.
    ELSEIF ls_prop-tbd = abap_true AND lv_exists = abap_true.
*     Never blank out a value someone already maintained
      ls_out-icon   = icon_led_inactive.
      ls_out-action = 'Kept - already exists'.
    ELSEIF lv_exists = abap_true AND
           ( p_overwr = abap_false OR ls_existing-z_output = ls_prop-z_output ).
      ls_out-icon   = icon_led_inactive.
      ls_out-action = 'Kept - already exists'.
    ELSE.
      ls_out-icon   = COND #( WHEN ls_prop-tbd = abap_true THEN icon_led_yellow ELSE icon_led_green ).
      ls_out-action = COND #( WHEN lv_exists = abap_true      THEN 'Updated'
                              WHEN ls_prop-tbd = abap_true    THEN 'Created empty - complete it'
                              ELSE 'Created' ).
      APPEND VALUE #( bukrs            = p_bukrs
                      field_name       = ls_prop-field_name
                      file_column_name = ls_prop-file_column_name
                      z_input          = ls_prop-z_input
                      z_output         = ls_prop-z_output ) TO lt_write.
    ENDIF.
    IF p_test = abap_true AND ls_out-action(4) <> 'Kept' AND ls_out-action(4) <> 'Skip'.
      ls_out-action = |(Test) { ls_out-action }|.
    ENDIF.
    APPEND ls_out TO gt_out.
  ENDLOOP.

  IF p_test = abap_true OR lt_write IS INITIAL.
    RETURN.
  ENDIF.

  MODIFY zsd_br_nfse_parm FROM TABLE @lt_write.
  IF sy-subrc = 0.
    COMMIT WORK.
    lv_count = lines( lt_write ).
    CONDENSE lv_count.
    MESSAGE |{ lv_count } { TEXT-s01 }| TYPE 'S'.          " entries written
  ELSE.
    ROLLBACK WORK.
    MESSAGE TEXT-e03 TYPE 'S' DISPLAY LIKE 'E'.             " nothing saved
    LOOP AT gt_out ASSIGNING FIELD-SYMBOL(<ls_out>) WHERE icon = icon_led_green.
      <ls_out>-icon   = icon_led_red.
      <ls_out>-action = 'Not saved - database error'.
    ENDLOOP.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  DISPLAY
*&---------------------------------------------------------------------*
FORM display.

  DATA lo_alv TYPE REF TO cl_salv_table.

  TRY.
      cl_salv_table=>factory( IMPORTING r_salv_table = lo_alv
                              CHANGING  t_table      = gt_out ).
    CATCH cx_salv_msg.
      RETURN.
  ENDTRY.

  lo_alv->get_functions( )->set_all( abap_true ).
  lo_alv->get_columns( )->set_optimize( abap_true ).
  lo_alv->get_display_settings( )->set_list_header(
    COND #( WHEN p_test = abap_true THEN |{ gc_table } { p_bukrs } - test run|
            ELSE |{ gc_table } { p_bukrs } - update run| ) ).

  TRY.
      DATA(lo_columns) = lo_alv->get_columns( ).
      lo_columns->get_column( 'ICON' )->set_short_text( 'Status' ).
      lo_columns->get_column( 'OLD_OUTPUT' )->set_short_text( 'Old value' ).
      lo_columns->get_column( 'OLD_OUTPUT' )->set_medium_text( 'Value before' ).
      lo_columns->get_column( 'OLD_OUTPUT' )->set_long_text( 'Value before' ).
      lo_columns->get_column( 'ACTION' )->set_short_text( 'Action' ).
      lo_columns->get_column( 'ACTION' )->set_medium_text( 'Action' ).
      lo_columns->get_column( 'ACTION' )->set_long_text( 'Action' ).
      lo_columns->get_column( 'NOTE' )->set_short_text( 'Note' ).
      lo_columns->get_column( 'NOTE' )->set_medium_text( 'Note' ).
      lo_columns->get_column( 'NOTE' )->set_long_text( 'Note' ).
    CATCH cx_salv_not_found.
  ENDTRY.

  lo_alv->display( ).

ENDFORM.
