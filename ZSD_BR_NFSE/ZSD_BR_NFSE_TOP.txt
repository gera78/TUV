TYPES:
* One expected Excel column: letter, header text, target field
  BEGIN OF gty_layout,
    column TYPE string,          " component name in the FDT itab (A, B, ..., BB)
    header TYPE string,          " expected header text (row 2)
    field  TYPE fieldname,       " target field in ZSD_BR_NFSE_DATA (initial = ignored)
  END OF gty_layout,
  gty_layout_tab TYPE STANDARD TABLE OF gty_layout WITH DEFAULT KEY,
  gty_fieldname_tab TYPE STANDARD TABLE OF fieldname WITH DEFAULT KEY,

  gty_data_tab  TYPE STANDARD TABLE OF zsd_br_nfse_data WITH DEFAULT KEY,
  gty_log_tab   TYPE STANDARD TABLE OF zsd_br_nfse_log WITH DEFAULT KEY,
  gty_file_tab  TYPE STANDARD TABLE OF zsd_br_nfse_file WITH DEFAULT KEY,
  gty_param_tab TYPE SORTED TABLE OF zsd_br_nfse_parm
                WITH UNIQUE KEY bukrs field_name file_column_name input,

* Next free log item per line, so new messages never collide with old ones
  BEGIN OF gty_log_counter,
    file_id  TYPE zsd_br_nfse_file_id,
    line_id  TYPE zsd_br_nfse_line_id,
    log_item TYPE zsd_br_nfse_log_item,
  END OF gty_log_counter,
  gty_log_counter_tab TYPE HASHED TABLE OF gty_log_counter
                      WITH UNIQUE KEY file_id line_id,

* Local class: ALV event handler (implementation in ZSD_BR_NFSE_ALV)
*---------------------------------------------------------------------*
CLASS lcl_alv_events DEFINITION FINAL.
  PUBLIC SECTION.
    CLASS-METHODS:
      on_link_click FOR EVENT link_click OF cl_salv_events_table
        IMPORTING row column,
      on_double_click FOR EVENT double_click OF cl_salv_events_table
        IMPORTING row column,
      on_user_command FOR EVENT added_function OF cl_salv_events
        IMPORTING e_salv_function.
ENDCLASS.
